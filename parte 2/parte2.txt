# 1. FUNCION PARA OBTENER COLABORADORES DE UN REPOSITORIO
def obtener_colaboradores(propietario, repositorio, max_usuarios=10):
    url = f'https://api.github.com/repos/{propietario}/{repositorio}/contributors?per_page={max_usuarios}'
    try:
        respuesta = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT)
        if respuesta.ok:
            return [usuario['login'] for usuario in respuesta.json()]
        print(f"Error al obtener colaboradores: {respuesta.status_code}")
        return []
    except Exception as e:
        print(f"Error de conexion para {propietario}/{repositorio}: {str(e)}")
        return []

# 2. FUNCION PARA OBTENER COMMITS REALES POR HORA
def obtener_commits_por_hora(propietario, repositorio, dias_retroceso=30):
    url = f'https://api.github.com/repos/{propietario}/{repositorio}/commits'
    fecha_inicio = (datetime.now() - timedelta(days=dias_retroceso)).isoformat()
    parametros = {'since': fecha_inicio, 'per_page': 100}
    
    datos_commits = []
    pagina = 1
    
    while True:
        try:
            parametros['page'] = pagina
            respuesta = requests.get(url, headers=headers, params=parametros, timeout=REQUEST_TIMEOUT)
            respuesta.raise_for_status()
            
            commits = respuesta.json()
            if not commits:
                break
                
            for commit in commits:
                if isinstance(commit, dict) and 'commit' in commit:
                    fecha_commit = datetime.strptime(
                        commit['commit']['committer']['date'],
                        '%Y-%m-%dT%H:%M:%SZ'
                    )
                    hora = fecha_commit.hour
                    dia_semana = fecha_commit.strftime('%A')
                    datos_commits.append({
                        'Día': dia_semana,
                        'Hora': hora,
                        'Fecha': fecha_commit.date()
                    })
            
            if 'next' not in respuesta.links:
                break
                
            pagina += 1
            time.sleep(0.5)
            
        except Exception as e:
            print(f"Error al obtener commits: {str(e)}")
            break
    
    return datos_commits

# Funcion para crear mapa de calor con commits reales
def crear_mapa_calor_real(propietario, repositorio):
    print(f"\n📊 Obteniendo commits REALES de {propietario}/{repositorio}...")
    
    datos_commits = obtener_commits_por_hora(propietario, repositorio)
    
    if not datos_commits:
        print("No se pudieron obtener commits. Usando datos de ejemplo...")
        return crear_mapa_calor_ejemplo(propietario, repositorio)
    
    df = pd.DataFrame(datos_commits)
    conteo_commits = df.groupby(['Día', 'Hora']).size().reset_index(name='Commits')
    
    orden_dias = ['Lunes', 'Martes', 'Miercoles', 'Jueves', 'Viernes', 'Sabado', 'Domingo']
    conteo_commits['Día'] = pd.Categorical(conteo_commits['Día'], categories=orden_dias, ordered=True)
    
    figura = px.density_heatmap(
        conteo_commits,
        x='Hora',
        y='Día',
        z='Commits',
        nbinsx=24,
        nbinsy=7,
        color_continuous_scale='thermal',
        title=f'Actividad REAL de Commits en {propietario}/{repositorio}',
        labels={'Commits': 'Commits/hora'}
    )
    
    figura.update_layout(
        xaxis=dict(
            tickvals=list(range(24)),
            ticktext=[f"{h:02d}:00" for h in range(24)],
            title='Hora del día (UTC)'
        ),
        yaxis=dict(title='Día de la semana'),
        coloraxis=dict(
            colorbar=dict(title='Commits/hora'),
            cmin=0,
            cmax=conteo_commits['Commits'].quantile(0.9)  
        ),
        height=600,
        hovermode='x unified'
    )
    
    return figura

# Funcion para crear mapa de calor con datos de ejemplo
def crear_mapa_calor_ejemplo(propietario, repositorio):
    print("Generando datos de ejemplo realistas...")
    actividad = []
    for dia in range(7):
        for hora in range(24):
            base = 5 if dia >= 5 else 10
            if dia < 5:  # Dias laborables
                if 9 <= hora <= 17:  # Horario laboral
                    commits = base + 30 * (1 - abs(13 - hora)/4)
                elif 19 <= hora <= 23:  # Tarde-noche
                    commits = base + 15
                else:  # Madrugada
                    commits = base
            else:  # Fin de semana
                if 10 <= hora <= 16:  # Horario diurno
                    commits = base + 20 * (1 - abs(13 - hora)/3)
                else:  # Noche/madrugada
                    commits = base
            commits += random.randint(0, 10)  # Variabilidad aleatoria
            actividad.append([hora, dia, int(max(commits, 1))])
    
    df = pd.DataFrame(actividad, columns=['Hora', 'Día_num', 'Commits'])
    dias = ['Lunes', 'Martes', 'Miercoles', 'Jueves', 'Viernes', 'Sabado', 'Domingo']
    df['Día'] = df['Día_num'].map(lambda x: dias[x])
    
    figura = px.density_heatmap(
        df,
        x='Hora',
        y='Día',
        z='Commits',
        color_continuous_scale='thermal',
        title=f'Actividad de Commits en {propietario}/{repositorio} (Datos de ejemplo)'
    )
    
    figura.add_annotation(
        text="⚠️ DATOS DE EJEMPLO - API NO RESPONDIO pipipi :( ⚠️",
        xref="paper", yref="paper",
        x=0.5, y=1.08,
        showarrow=False,
        font=dict(size=12, color="red")
    )
    
    return figura

# Crear mapa de calor para React
figura_mapa_calor = crear_mapa_calor_real('facebook', 'react')

# 3. FUNCION PARA OBTENER REPOSITORIOS POPULARES CON LICENCIAS
def obtener_repositorios_populares(num_repos=50):
    url = f'https://api.github.com/search/repositories?q=stars:>1000&sort=stars&order=desc&per_page={num_repos}'
    try:
        respuesta = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT)
        if respuesta.ok:
            repositorios = respuesta.json().get('items', [])
            datos = []
            for repo in repositorios:
                try:
                    info_licencia = repo.get('license', {})
                    datos.append({
                        'nombre': repo.get('name', 'Desconocido'),
                        'lenguaje': repo.get('language', 'None'),
                        'licencia': info_licencia.get('spdx_id', 'None') if info_licencia else 'None',
                        'estrellas': repo.get('stargazers_count', 0)
                    })
                except Exception as e:
                    print(f"Error procesando repositorio: {str(e)}")
                    continue
            return pd.DataFrame(datos)
        print(f"Error al obtener repositorios: {respuesta.status_code}")
        return pd.DataFrame()
    except Exception as e:
        print(f"Error de conexión: {str(e)}")
        return pd.DataFrame()
