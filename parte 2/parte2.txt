def get_collaborators(owner, repo, max_users=10):

    try:
        response = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT)
        if response.ok:
            return [user['login'] for user in response.json()]
        print(f"Error al obtener colaboradores: {response.status_code}")
        return []
    except Exception as e:
        print(f"Error de conexión para {owner}/{repo}: {str(e)}")
        return []

# --- 2. MAPA DE CALOR CON COMMITS REALES POR HORA ---
def get_real_commits_by_hour(owner, repo, days_back=30):
    """Obtiene commits reales agrupados por hora y día"""
    url = f'https://api.github.com/repos/{owner}/{repo}/commits'
    since_date = (datetime.now() - timedelta(days=days_back)).isoformat()
    params = {'since': since_date, 'per_page': 100}
    
    commits_data = []
    page = 1
    
    while True:
        try:
            params['page'] = page
            response = requests.get(url, headers=headers, params=params, timeout=REQUEST_TIMEOUT)
            response.raise_for_status()
            
            commits = response.json()
            if not commits:
                break
                
            for commit in commits:
                if isinstance(commit, dict) and 'commit' in commit:
                    commit_date = datetime.strptime(
                        commit['commit']['committer']['date'],
                        '%Y-%m-%dT%H:%M:%SZ'
                    )
                    hour = commit_date.hour
                    weekday = commit_date.strftime('%A')
                    commits_data.append({
                        'Día': weekday,
                        'Hora': hour,
                        'Fecha': commit_date.date()
                    })
            
            # Verificar si hay más páginas
            if 'next' not in response.links:
                break
                
            page += 1
            time.sleep(0.5)  # Evitar rate limiting
            
        except Exception as e:
            print(f"Error al obtener commits: {str(e)}")
            break
    
    return commits_data

def create_real_heatmap(owner, repo):
    print(f"\n📊 Obteniendo commits REALES de {owner}/{repo}...")
    
    commits_data = get_real_commits_by_hour(owner, repo)
    
    if not commits_data:
        print("No se pudieron obtener commits. Usando datos de ejemplo...")
        return create_sample_heatmap(owner, repo)
    
    # Procesamiento de datos
    df = pd.DataFrame(commits_data)
    df_counts = df.groupby(['Día', 'Hora']).size().reset_index(name='Commits')
    
    # Ordenar días lógicamente
    days_order = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']
    df_counts['Día'] = pd.Categorical(df_counts['Día'], categories=days_order, ordered=True)
    
    # Crear heatmap
    fig = px.density_heatmap(
        df_counts,
        x='Hora',
        y='Día',
        z='Commits',
        nbinsx=24,
        nbinsy=7,
        color_continuous_scale='thermal',
        title=f'Actividad REAL de Commits en {owner}/{repo}',
        labels={'Commits': 'Commits/hora'}
    )
    
    # Personalización
    fig.update_layout(
        xaxis=dict(
            tickvals=list(range(24)),
            ticktext=[f"{h:02d}:00" for h in range(24)],
            title='Hora del día (UTC)'
        ),
        yaxis=dict(title='Día de la semana'),
        coloraxis=dict(
            colorbar=dict(title='Commits/hora'),
            cmin=0,
            cmax=df_counts['Commits'].quantile(0.9)  # Escala al percentil 90
        ),
        height=600,
        hovermode='x unified'
    )
    
    return fig

def create_sample_heatmap(owner, repo):
    """Genera datos de ejemplo cuando la API falla"""
    print("Generando datos de ejemplo realistas...")
    activity = []
    for day in range(7):
        for hour in range(24):
            base = 5 if day >= 5 else 10
            if day < 5:
                if 9 <= hour <= 17:
                    commits = base + 30 * (1 - abs(13 - hour)/4)
                elif 19 <= hour <= 23:
                    commits = base + 15
                else:
                    commits = base
            else:
                if 10 <= hour <= 16:
                    commits = base + 20 * (1 - abs(13 - hour)/3)
                else:
                    commits = base
            commits += random.randint(0, 10)
            activity.append([hour, day, int(max(commits, 1))])
    
    df = pd.DataFrame(activity, columns=['Hora', 'Día_num', 'Commits'])
    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']
    df['Día'] = df['Día_num'].map(lambda x: days[x])
    
    fig = px.density_heatmap(
        df,
        x='Hora',
        y='Día',
        z='Commits',
        color_continuous_scale='thermal',
        title=f'Actividad de Commits en {owner}/{repo} (Datos de ejemplo)'
    )
    
    fig.add_annotation(
        text="⚠️ DATOS DE EJEMPLO - API NO RESPONDIÓ ⚠️",
        xref="paper", yref="paper",
        x=0.5, y=1.08,
        showarrow=False,
        font=dict(size=12, color="red")
    )
    
    return fig

# --- 3. TREEMAP DE LICENCIAS Y LENGUAJES ---
def get_top_repos_with_licenses(num_repos=50):
    url = f'https://api.github.com/search/repositories?q=stars:>1000&sort=stars&order=desc&per_page={num_repos}'
    try:
        response = requests.get(url, headers=headers, timeout=REQUEST_TIMEOUT)
        if response.ok:
            repos = response.json().get('items', [])
            data = []
            for repo in repos:
                try:
                    license_info = repo.get('license', {})
                    data.append({
                        'name': repo.get('name', 'Desconocido'),
                        'language': repo.get('language', 'None'),
                        'license': license_info.get('spdx_id', 'None') if license_info else 'None',
                        'stars': repo.get('stargazers_count', 0)
                    })
                except Exception as e:
                    print(f"Error procesando repositorio: {str(e)}")
                    continue
            return pd.DataFrame(data)
        print(f"Error al obtener repositorios: {response.status_code}")
        return pd.DataFrame()
    except Exception as e:
        print(f"Error de conexión: {str(e)}")
        return pd.DataFrame()

df_licenses = get_top_repos_with_licenses()

try:
    fig_treemap = px.treemap(
        df_licenses,
        path=['license', 'language', 'name'],
        values='stars',
        title='Distribución de Licencias y Lenguajes en Repositorios Populares',
        color_continuous_scale='RdBu'
    ) if not df_licenses.empty else go.Figure()
except Exception as e:
    print(f"Error al crear treemap: {str(e)}")
    fig_treemap = go.Figure()
